## 🔁 페이징 처리 로직 개요

- **조건 1**: MCI 기준 전문 길이가 10K 미만이 되도록 건수 분할
    - → MCI 항목별 길이 합산하여 계산
    - → 일반 조회: 100건, 데이터 크기 클 경우: 50건 단위로 나눔
- **조건 2**: 다음 페이지 존재 시 최대 **5회 반복 조회**
- **조건 3**: 5회마다 사용자에게 계속 조회할 것인지 확인

## 📦 조회 스크립트 (정리 및 주석 추가)

```javascript
// 📌 조회 횟수 및 설정
var intRetrieve = 1;       // 현재 조회 페이지 수
var intTranTerm = 5;       // 몇 번째마다 사용자 확인 메시지 띄울지

// 📌 그리드 및 변수 초기화
function initObjData() {
  edx_PAGE_NO.Data = 1;
  edx_PAGE_SIZE.Data = 50;              // 기본 50건 (전문 길이 기준 상황에 따라 100으로 변경 가능)
  edx_GRID_BBO_ROWCOUNT.Data = 0;
  edx_NEXT_PAGE_EXIS_YN.Data = "N";
  grd_GRID_BBO.RemoveAll();
}

// 📌 조회 요청
function retrieveList(vType) {
  if (vType == "First") {
    initObjData();
    intRetrieve = 1;
    grd_GRID_BBO.AppendMode = false;
  } else {
    intRetrieve++;
    grd_GRID_BBO.AppendMode = true;
  }

  edx_PAGE_NO.Data = intRetrieve + "";

  SetTranID("BBO00000011");  // 거래 ID 설정
  Tran.ASend;                // 거래 실행
}

// 📌 초기화 이벤트
function OnInitScreen() {
  initObjData();
}

// 📌 거래 완료 이벤트
function OnEndTran() {
  if (Tran.bError == false) {

    // 다음 페이지 존재 시 처리
    if (edx_NEXT_PAGE_EXIS_YN.Data == "Y") {

      // 5번째 마다 사용자에게 메시지
      if ((intRetrieve % intTranTerm) == 0) {
        var m = MessageBox("다음 데이터가 존재합니다. 거래를 계속 하시겠습니까?", "후속 데이터 출력 확인", 0);
        if (m == 6) { // Yes 선택 시
          retrieveList("N");
        }
      } else {
        retrieveList("N"); // 자동 조회
      }

    } else {
      // 다음 페이지 없음
      edx_PAGE_NO.Data = 1;
      grd_GRID_BBO.AppendMode = false;
      grd_GRID_BBO.OnClick(1, 1, 1);  // 첫 행 클릭
    }
  }

  if (grd_GRID_BBO.RowCount == 0) {
    Alert("해당 데이터가 없습니다", "");
  }
}

// 📌 조회 버튼 클릭 시 이벤트
function btn_QRY_OnClick() {
  grd_GRID_BBO.RowCount = 0;
  grd_GRID_BBO.Clear();
  retrieveList("First");
}
```
